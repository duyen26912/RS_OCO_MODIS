Bước 1: Cắt ranh giới Việt Nam bởi vì vùng Lancover rộng hơn 
gdalwarp -cutline "E:\RanhGioi\VNM_adm\gadm41_VNM_0.shp" ^
 -crop_to_cutline ^
 -multi ^
 -wo NUM_THREADS=4 ^
 -wm 2048 ^
 --config GDAL_CACHEMAX 1024 ^
 -co COMPRESS=LZW ^
 -co TILED=YES ^
 -co BIGTIFF=YES ^
 landcover_uncut_10m.tif lc_vn_10m_wgs84.tif

Bước 2: Chuyển đổi các pixel == 3 thành band 1 và các pixel khác thành band 0
Trong file landcover_convert_rice_binary.ipynb

Bước 3: Chuyển đổi hệ tọa độ từ wgs84 4326 sang utm48N
gdalwarp -t_srs EPSG:32648 -r near ^
  -co COMPRESS=LZW ^
  -co TILED=YES ^
  -co BIGTIFF=YES ^
  -multi ^
  -wo NUM_THREADS=4 ^
  --config GDAL_CACHEMAX 512 ^
  lc_vn_10m_wgs84_rice_binary.tif lc_vn_10m_utm_rice_binary.tif

Bước 4:  Chuyển từ 10m → 250m
gdalwarp -tr 250 250 -r mode ^
  -co COMPRESS=LZW ^
  -co TILED=YES ^
  lc_vn_10m_utm_rice_binary.tif lc_vn_250m_utm_rice_binary.tif
